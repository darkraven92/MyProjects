cmake_minimum_required(VERSION 3.16)
project(svearike C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(svea_core src/director.c src/trade.c src/assets.c)
target_include_directories(svea_core PUBLIC src)

  add_compile_options(-Wall -Wextra -Wpedantic)
  target_compile_options(svea_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(svea_assets PRIVATE -Wall -Wextra -Wpedantic)
endif()
